Google Sheets Sync
📱
Mobile tip: Do steps 1–3 once on a desktop (~3 min). The same URL is used for both data storage AND live prices via Google Finance.
1Create Google Sheet
Go to sheets.google.com → New sheet → Extensions → Apps Script
2Paste script & save
Replace everything with the code below → Ctrl+S (includes GOOGLEFINANCE price fetcher)
const SHEET='Holdings'; const H=['id','name','type','symbol','units','buyPrice', 'buyDate','currency','notes','soldPrice','soldDate','soldNotes','status']; // ── Holdings CRUD ────────────────────────────────────────────────────────── function doGet(e) { var params = e ? (e.parameter || {}) : {}; if (params.action === 'prices') { return getPrices(params.symbols || ''); } var sh = getSheet(); if (sh.getLastRow() === 0) sh.appendRow(H); var range = sh.getDataRange().getValues(); var headers = range[0]; var rows = range.slice(1); var data = rows.map(function(r) { var obj = {}; headers.forEach(function(h, i) { obj[h] = r[i]; }); return obj; }); return out({ data: data }); } function doPost(e) { var b = JSON.parse(e.postData.contents); var sh = getSheet(); if (sh.getLastRow() === 0) sh.appendRow(H); if (b.action === 'add') { sh.appendRow(H.map(function(h) { return b[h] !== undefined ? b[h] : ''; })); } else { var vals = sh.getDataRange().getValues(); for (var i = 1; i < vals.length; i++) { if (String(vals[i][0]) === String(b.id)) { if (b.action === 'delete') { sh.deleteRow(i + 1); } else { sh.getRange(i + 1, 1, 1, H.length) .setValues([H.map(function(h) { return b[h] !== undefined ? b[h] : ''; })]); } break; } } } return out({ ok: true }); } // ── Price Fetcher using UrlFetchApp (server-side, no CORS) ───────────────── // Called as: GET ?action=prices&symbols=nse:RELIANCE|us:AAPL|mf:120503|gold:|silver:|fx: // Returns: { prices: { "nse:RELIANCE": { price, prev }, ... }, usdInr: 84 } function getPrices(symbolsParam) { var symbols = symbolsParam.split('|').filter(function(s) { return s.length > 0; }); var prices = {}; var usdInr = 84; // First fetch USD/INR (needed for gold, silver, US stocks) try { var fxData = yahooQuote('USDINR=X'); if (fxData && fxData.price > 50 && fxData.price < 200) { usdInr = fxData.price; } } catch(e) {} var ozToGram = usdInr / 31.1035; symbols.forEach(function(raw) { try { var parts = raw.split(':'); var type = parts[0].toLowerCase(); var sym = parts[1] || ''; if (type === 'mutual_fund' || type === 'mf') { // MFAPI — official AMFI NAV data var resp = UrlFetchApp.fetch('https://api.mfapi.in/mf/' + sym, { muteHttpExceptions: true }); if (resp.getResponseCode() === 200) { var j = JSON.parse(resp.getContentText()); var nav0 = parseFloat(j.data[0].nav); var nav1 = parseFloat((j.data[1] || j.data[0]).nav); if (nav0 > 0) { prices[raw] = { price: nav0, prev: nav1, src: 'MFAPI' }; } } } else if (type === 'gold') { // MCX Gold futures (GOLD.MCX) = ₹ per 10 grams on MCX India // Divide by 10 to get ₹ per gram var g = yahooQuote('GOLD.MCX'); if (g && g.price > 1000) { prices[raw] = { price: Math.round(g.price / 10 * 100) / 100, prev: Math.round(g.prev / 10 * 100) / 100, src: 'MCX' }; } else { // Fallback: COMEX GC=F converted var gc = yahooQuote('GC=F'); if (gc) prices[raw] = { price: Math.round(gc.price * ozToGram * 100) / 100, prev: Math.round(gc.prev * ozToGram * 100) / 100, src: 'COMEX' }; } } else if (type === 'silver') { // MCX Silver futures (SILVER.MCX) = ₹ per kg on MCX India // Divide by 1000 to get ₹ per gram var si = yahooQuote('SILVER.MCX'); if (si && si.price > 10000) { prices[raw] = { price: Math.round(si.price / 1000 * 100) / 100, prev: Math.round(si.prev / 1000 * 100) / 100, src: 'MCX' }; } else { // Fallback: COMEX SI=F converted var sic = yahooQuote('SI=F'); if (sic) prices[raw] = { price: Math.round(sic.price * ozToGram * 100) / 100, prev: Math.round(sic.prev * ozToGram * 100) / 100, src: 'COMEX' }; } } else if (type === 'nse') { var q = yahooQuote(sym + '.NS'); if (q) prices[raw] = { price: q.price, prev: q.prev, src: 'YF' }; } else if (type === 'bse') { var q = yahooQuote(sym + '.BO'); if (q) prices[raw] = { price: q.price, prev: q.prev, src: 'YF' }; } else if (type === 'us') { var q = yahooQuote(sym); if (q) prices[raw] = { price: q.price, prev: q.prev, src: 'YF' }; } } catch(ex) { // silently skip — app falls back to client-side Yahoo } }); return out({ prices: prices, usdInr: usdInr }); } // Yahoo Finance v7 — called server-side via UrlFetchApp (no CORS issue) function yahooQuote(symbol) { var url = 'https://query2.finance.yahoo.com/v7/finance/quote?symbols=' + encodeURIComponent(symbol) + '&fields=regularMarketPrice,regularMarketPreviousClose'; var opts = { muteHttpExceptions: true, headers: { 'User-Agent': 'Mozilla/5.0', 'Accept': 'application/json' } }; var resp = UrlFetchApp.fetch(url, opts); if (resp.getResponseCode() !== 200) return null; var j = JSON.parse(resp.getContentText()); var r = j && j.quoteResponse && j.quoteResponse.result && j.quoteResponse.result[0]; if (!r || !r.regularMarketPrice || r.regularMarketPrice <= 0) return null; return { price: r.regularMarketPrice, prev: r.regularMarketPreviousClose || r.regularMarketPrice }; } function getSheet() { return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET) || SpreadsheetApp.getActiveSpreadsheet().insertSheet(SHEET); } function out(o) { var r = ContentService.createTextOutput(JSON.stringify(o)); r.setMimeType(ContentService.MimeType.JSON); return r; }
